import Tabs from "@theme/Tabs";
import TabItem from "@theme/TabItem";
import CodeBlock from "@theme/CodeBlock";
import useDocusaurusContext from "@docusaurus/useDocusaurusContext";
export const customFields = () => useDocusaurusContext().siteConfig.customFields;
export const siteConfig = () => useDocusaurusContext().siteConfig;
export const PowerShellInstall = () => {
  const siteConfig = useDocusaurusContext().siteConfig;
  const docsUrl = `${siteConfig.url}${siteConfig.baseUrl}`.replace(/\/$/, "");
  return (
    <CodeBlock language="powershell">{`Set-ExecutionPolicy Bypass -Scope Process -Force; Invoke-Expression ((New-Object System.Net.WebClient).DownloadString('${docsUrl}/install.ps1'))`}</CodeBlock>
  );
}
