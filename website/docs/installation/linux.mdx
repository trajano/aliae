import Tabs from "@theme/Tabs";
import TabItem from "@theme/TabItem";
import InstallHomebrew from "./homebrew.mdx";
import CodeBlock from "@theme/CodeBlock";
import useDocusaurusContext from "@docusaurus/useDocusaurusContext";
export const customFields = () =>
  useDocusaurusContext().siteConfig.customFields;
export const BashInstall = ({ binDir = false }) => {
  const siteConfig = useDocusaurusContext().siteConfig;
  const docsUrl = `${siteConfig.url}${siteConfig.baseUrl}`.replace(/\/$/, "");
  if (binDir) {
    return (
      <CodeBlock language="bash">{`curl -s ${docsUrl}/install.sh | bash -s`}</CodeBlock>
    );
  } else {
    return (
      <CodeBlock language="bash">{`curl -s ${docsUrl}/install.sh | bash -s -- -d ~/bin`}</CodeBlock>
    );
  }
};

If you want to install to a different location you can specify it using the `-d` flag:
